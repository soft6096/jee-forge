### #2 POST /api/v1/users/detail — 用户详情

**入参 JSON**：

```json
{
  "id": 10001 // 【必填】Long，用户ID（缺失 → 500「用户主键不能为空」）
}
```

**出参 JSON（成功 / 失败）**：

```json
// 成功（200）
{
  "code": 200, // 【状态码】Integer，200=成功
  "message": "操作成功", // 【提示信息】String
  "data": { // 【业务数据】Object，用户详情
    "id": 10001, // 用户ID（Long）
    "username": "zhangsan", // 用户名（String）
    "email": "zhangsan@example.com", // 邮箱（String）
    "deptId": 101, // 部门ID（Long）
    "deptName": "研发部", // 部门名称（String，后端转义，前端直接展示）
    "status": "0", // 状态（字典 <字典类型码>）：0-正常 / 1-停用
    "statusName": "正常", // 状态中文名（String，后端转义）
    "version": 1 // 【选填】乐观锁版本号（Integer）：编辑接口需**原样回传**；无乐观锁的项目删掉本行
  },
  "timestamp": "2026-09-29 20:06:00", // 【时间戳】String
  "traceId": "d4e5f6a7b8c9" // 【链路ID】String
}
// 失败（500 用户不存在 / 已逻辑删除）
{
  "code": 500, // 500=<后端异常枚举名，如 USER_NOT_EXIST>
  "message": "用户不存在", // 【提示信息】String，后端异常枚举原文
  "timestamp": "2026-09-29 20:06:00", // 【时间戳】String
  "traceId": "d4e5f6a7b8c9" // 【链路ID】String
}
```

### #3 POST /api/v1/users/page — 分页查询用户

**入参 JSON**（分页字段继承基类，如 `BasePageReq`）：

```json
{
  "username": "zhang", // 【选填】String，用户名（模糊匹配）
  "statusList": ["0", "1"], // 【选填】Array<String>，状态多选（字典 <字典类型码>）；不传=不筛选
  "createTimeStart": "2026-09-01 00:00:00", // 【选填】Date，创建开始时间（含），格式 yyyy-MM-dd HH:mm:ss
  "createTimeEnd": "2026-09-29 23:59:59", // 【选填】Date，创建结束时间（含），格式 yyyy-MM-dd HH:mm:ss
  "currentPage": 1, // 【选填】Long，当前页码，从 1 开始（继承基类，默认 1）
  "pageSize": 10 // 【选填】Long，每页条数（继承基类，默认 10；上限 100，超出静默截断为 100）
}
```

**出参 JSON（成功 / 失败）**：

```json
// 成功（200）
{
  "code": 200, // 【状态码】Integer，200=成功
  "message": "操作成功", // 【提示信息】String
  "data": { // 【业务数据】Object，分页结构（框架 `Page`）
    "records": [ // 当前页数据列表（Array）
      {
        "id": 10001, // 用户ID（Long）
        "username": "zhangsan", // 用户名（String）
        "status": "0", // 状态（字典 <字典类型码>）
        "statusName": "正常", // 状态中文名（String）
        "createTime": "2026-09-29 10:00:00" // 创建时间（Date，yyyy-MM-dd HH:mm:ss）
      }
    ],
    "total": 1, // 总记录数（Long）
    "size": 10, // 每页条数（Long；被截断时为实际值）
    "current": 1, // 当前页码（Long）
    "pages": 1 // 总页数（Long）
  },
  "timestamp": "2026-09-29 20:07:00", // 【时间戳】String
  "traceId": "e5f6a7b8c9d0" // 【链路ID】String
}
// 成功但无数据（200）—— 列表为空不是失败，records 给空数组
{
  "code": 200, // 200=成功（前端展示空态，不要当异常处理）
  "message": "操作成功", // 【提示信息】String
  "data": { "records": [], "total": 0, "size": 10, "current": 1, "pages": 0 }, // records 空数组 / total 0 / pages 0
  "timestamp": "2026-09-29 20:07:00", // 【时间戳】String
  "traceId": "e5f6a7b8c9d0" // 【链路ID】String
}
```
