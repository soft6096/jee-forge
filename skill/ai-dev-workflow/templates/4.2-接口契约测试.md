# 契约测试骨架

> 使用说明：这是第 ④ 步"契约测试"的产物模板。从技术方案的"验收场景"表直接翻译成测试。
> 核心规则：
> 1. **测试由你写**（或让 AI 生成骨架、你审核填充数据）——不能交给实现代码的同一个 AI，否则会"骗绿"
> 2. 每个接口至少两个用例：合法入参→期望出参；非法入参→错误码
> 3. **测试方法名英文驼峰**（`createOrderShouldReturn400WhenProductNotExist`），禁止中文/拼音方法名；用例场景中文写方法 Javadoc（见下方示例）
> 3. 断言必须具体：断字段值、错误码，禁止只断"返回 200"
> 4. 写完后**先跑一遍确认红**，再进入第 ⑤ 步
> 示例为**订单创建接口**，与《3.0-技术方案-通用骨架》验收场景章节 + 《3.1-技术方案-Controller》示例中的验收场景一一对应。

## Controller 契约测试（MockMvc 示例）

```java
package com.example.order.controller;

import com.example.order.domain.dto.OrderDTO;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.AutoConfigureMockMvc;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import java.math.BigDecimal;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class OrderControllerTest {

    @Autowired
    private MockMvc mockMvc;

    @Autowired
    private ObjectMapper objectMapper;

    /** 场景 #1：合法入参 → 200 + data.orderId 存在 */
    @Test
    void createOrderShouldReturnOrderIdWhenValid() throws Exception {
        OrderDTO dto = new OrderDTO();
        dto.setUserId(88L);
        dto.setItemIdList(java.util.List.of(1L, 2L));
        dto.setAmount(new BigDecimal("299.00"));
        dto.setRemark("加急配送");

        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(200))
            .andExpect(jsonPath("$.data.orderId").isNumber());   // 断具体字段，不是只断 200
    }

    /** 场景 #3：商品不存在 → 400 + 错误提示 */
    @Test
    void createOrderShouldReturn400WhenProductNotExist() throws Exception {
        OrderDTO dto = new OrderDTO();
        dto.setUserId(88L);
        dto.setItemIdList(java.util.List.of(99999L));   // 不存在的商品
        dto.setAmount(new BigDecimal("100.00"));

        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isBadRequest())
            .andExpect(jsonPath("$.code").value(400))
            .andExpect(jsonPath("$.msg").value("商品不存在"));
    }

    /** 场景 #4：金额非法（负数）→ 400 */
    @Test
    void createOrderShouldReturn400WhenAmountNegative() throws Exception {
        OrderDTO dto = new OrderDTO();
        dto.setUserId(88L);
        dto.setItemIdList(java.util.List.of(1L));
        dto.setAmount(new BigDecimal("-1.00"));

        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isBadRequest())
            .andExpect(jsonPath("$.code").value(400));
    }

    /** 场景 #7：未授权 → 403 */
    @Test
    void createOrderShouldReturn403WhenUnauthorized() throws Exception {
        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content("{}"))   // 不带 token
            .andExpect(status().isForbidden())
            .andExpect(jsonPath("$.code").value(403));
    }
}
```

## Listener 契约测试骨架（订单支付回调示例）

package com.example.order.mq;

import org.junit.jupiter.api.Test;
import org.springframework.amqp.core.Message;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

import static org.assertj.core.api.Assertions.assertThat;

@SpringBootTest
class OrderPayListenerTest {

    @Autowired
    private OrderPayListener listener;

    @Autowired
    private OrderMapper orderMapper;

    /** 场景 #1：正常消息 → 状态更新 + 日志写入 */
    @Test
    void consumeShouldMarkOrderPaid() throws Exception {
        String msg = "{\"orderId\":10001,\"status\":\"PAID\",\"timestamp\":\"2026-08-07T10:00:00+08:00\"}";

        listener.onMessage(buildMessage(msg));

        Order order = orderMapper.selectById(10001L);
        assertThat(order.getStatus()).isEqualTo("PAID");
    }

    /** 场景 #2：同一条消息发两次 → 幂等跳过 */
    @Test
    void consumeShouldSkipDuplicateMessage() throws Exception {
        String msg = "{\"orderId\":10001,\"status\":\"PAID\",\"timestamp\":\"2026-08-07T10:00:00+08:00\"}";

        listener.onMessage(buildMessage(msg));
        listener.onMessage(buildMessage(msg));   // 第二次应被幂等拦截

        assertThat(orderStatusLogMapper.countByOrderAndStatus(10001L, "PAID")).isEqualTo(1);
    }

    /** 场景 #3：非法消息体 → 丢弃 + 告警（不抛异常，不入死信） */
    @Test
    void consumeShouldDropInvalidMessage() throws Exception {
        String msg = "{\"orderId\":10001}";   // 缺 status

        listener.onMessage(buildMessage(msg));   // 应吞掉并告警，不抛异常

        // 断言：订单状态未变 + 有告警日志
    }

    private Message buildMessage(String json) {
        return new org.springframework.amqp.core.Message(json.getBytes(), null);
    }
}

    @Autowired

    @Test
    void runShouldCloseTimeoutOrders() throws Exception {
        job.run();   // 手动触发

        assertThat(orderMapper.countByStatus("CLOSED")).isGreaterThan(0);
    }

    /** 场景 #2：并发触发两次 → 只执行一次（锁生效） */
    @Test
    void runShouldExecuteOnceWhenConcurrent() throws Exception {
        // 用两个线程同时调 run()，断言 job_execution_log 只有一条记录

        Thread t2 = new Thread(() -> { try { job.run(); } catch (Exception ignored) {} });
        t1.start(); t2.start();
        t1.join(); t2.join();

        // 断言：同一秒内 execution log 只有 1 条
    }

    /** 场景 #4：单条失败不影响批量 */
    @Test
